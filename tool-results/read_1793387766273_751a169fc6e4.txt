    95→    95→  // ------------------------------------------------ heavy aggregates @1Hz
    96→    96→  useEffect(() => {
    97→    97→    const st = getStream()
    98→    98→    const tNow = now()
    99→    99→    const tMin = tNow - span
   100→   100→    const binSize = meta.bin * binMult
   101→   101→    const trades = st.trades
   102→   102→    const win = trades.length && trades[trades.length - 1].t >= tMin - 60_000
   103→   103→      ? trades.filter((x) => x.t >= tMin)
   104→   104→      : []
   105→   105→
   106→   106→    // traded volume per price bin + POC
   107→   107→    const pocMap = new Map<number, number>()
   108→   108→    let buyNot = 0, sellNot = 0
   109→   109→    const bubbles: Aggs['bubbles'] = []
   110→   110→    const dots: Aggs['dots'] = []
   111→   111→    for (const tr of win) {
   112→   112→      const n = tr.p * tr.q
   113→   113→      const buy = !tr.m
   114→   114→      const bucket = Math.round(tr.p / binSize) * binSize
   115→   115→      pocMap.set(bucket, (pocMap.get(bucket) ?? 0) + n)
   116→   116→      if (buy) buyNot += n; else sellNot += n
   117→   117→      if (n >= bubbleMin) bubbles.push({ t: tr.t, p: tr.p, n, buy })
   118→   118→      else if (n >= bubbleMin / 4) dots.push({ t: tr.t, p: tr.p, n, buy })
   119→   119→    }
   120→   120→    bubbles.sort((a, b) => a.n - b.n)
   121→   121→    if (bubbles.length > 1600) bubbles.splice(0, bubbles.length - 1600)
   122→   122→    if (dots.length > 1400) { // stride-sample small dots
   123→   123→      const stride = Math.ceil(dots.length / 1400)
   124→   124→      const kept = dots.filter((_, i) => i % stride === 0)
   125→   125→      dots.length = 0; dots.push(...kept)
   126→   126→    }
   127→   127→    let poc: number | null = null
   128→   128→    let bestV = 0
   129→   129→    for (const [p, v] of pocMap) if (v > bestV) { bestV = v; poc = p }
   130→   130→
   131→   131→    // decimated traded price line (≤ 500 points)
   132→   132→    const priceLine: Aggs['priceLine'] = []
   133→   133→    if (win.length > 500) {
   134→   134→      const stride = win.length / 500
   135→   135→      for (let i = 0; i < 500; i++) {
   136→   136→        const tr = win[Math.floor(i * stride)]
   137→   137→        priceLine.push({ t: tr.t, p: tr.p })
   138→   138→      }
   139→   139→      priceLine.push({ t: win[win.length - 1].t, p: win[win.length - 1].p })
   140→   140→    } else {
   141→   141→      for (const tr of win) priceLine.push({ t: tr.t, p: tr.p })
   142→   142→    }
   143→   143→
   144→   144→    // decision levels from walls
   145→   145→    const mid = st.heatCols.length ? st.heatCols[st.heatCols.length - 1].mid : 0
   146→   146→    let buyAbove: number | null = null
   147→   147→    let sellBelow: number | null = null
   148→   148→    for (const w of st.walls) {
   149→   149→      if (w.side === 'ask' && w.p > mid && (buyAbove === null || w.p < buyAbove)) buyAbove = w.p
   150→   150→      if (w.side === 'bid' && w.p < mid && (sellBelow === null || w.p > sellBelow)) sellBelow = w.p
   151→   151→    }
   152→   152→
   153→   153→    // intensity reference: median level notional of latest column
   154→   154→    let refQ = 1
   155→   155→    const last = st.heatCols[st.heatCols.length - 1]
   156→   156→    if (last) {
   157→   157→      const arr: number[] = []
   158→   158→      for (const lv of last.bids.slice(0, 120)) arr.push(lv.q * lv.p)
   159→   159→      for (const lv of last.asks.slice(0, 120)) arr.push(lv.q * lv.p)
   160→   160→      if (arr.length > 10) {
   161→   161→        arr.sort((a, b) => a - b)
   162→   162→        refQ = arr[Math.floor(arr.length / 2)] || 1
   163→   163→      }
   164→   164→    }
   165→   165→
   166→   166→    // window trade rate
   167→   167→    const dt = win.length > 1 ? (win[win.length - 1].t - win[0].t) / 1000 : 1
   168→   168→    const tradeRate = dt > 0 ? win.length / dt : 0
   169→   169→
   170→   170→    aggsRef.current = { poc, pocMap, bubbles, dots, netBubble: buyNot - sellNot, bubbleCount: bubbles.length + dots.length, priceLine, buyAbove, sellBelow, refQ, tradeRate }
   171→   171→  }, [beat, span, binMult, bubbleMin, symbol])

   173→   173→  // ------------------------------------------------ geometry helpers per frame
   174→   174→  function geom(plotW: number, plotH: number) {
   175→   175→    const st = getStream()
   176→   176→    const cols = st.heatCols
   177→   177→    const tNow = now()
   178→   178→    const tMin = tNow - span
   179→   179→    if (!cols.length) return null
   180→   180→    const midNow = cols[cols.length - 1].mid
   181→   181→    let anchor = anchorRef.current
   182→   182→    const windowH = midNow * (vspan / 100) * 2
   183→   183→    if (!anchor || Math.abs(midNow - anchor.mid) > windowH * 0.35) {
   184→   184→      anchor = { mid: midNow }
   185→   185→      anchorRef.current = anchor
   186→   186→      drawStateRef.current = { vTotal: 0, shifted: 0, rawDrawn: 0, lastColsLen: -1, acc: null }
   187→   187→    }
   188→   188→    const m = anchor.mid
   189→   189→    const pTop = m * (1 + (vspan * 1.12) / 100)
   190→   190→    const pBot = m * (1 - (vspan * 1.12) / 100)
   191→   191→    const binSize = meta.bin * binMult
   192→   192→    const maxCols = Math.min(900, Math.max(80, Math.floor(plotW / 1.35)))
   193→   193→    const g = Math.max(1, Math.ceil((span / 250) / maxCols)) // merge factor
   194→   194→    const colW = Math.max(1.15, plotW / maxCols)
   195→   195→    return { st, cols, tNow, tMin, anchor: m, pTop, pBot, binSize, maxCols, g, colW, midNow }
   196→   196→  }

   198→   198→  const p2yOf = (p: number, pTop: number, pBot: number, plotH: number) =>
   199→   199→    ((pTop - p) / (pTop - pBot)) * plotH

   201→   201→  // ------------------------------------------------ incremental depth render
   202→   202→  function renderDepthInto(hc: HTMLCanvasElement, G: ReturnType<typeof geom>, plotW: number, plotH: number) {
   203→   203→    if (!G) return
   204→   204→    const { cols, tMin, pTop, pBot, binSize, g, colW } = G
   205→   205→    const ctx = hc.getContext('2d')!
   206→   206→    const ds = drawStateRef.current
   207→   207→    if (!ds.acc) ds.acc = { rows: new Map(), n: 0 }
   208→   208→
   209→   209→    // only recompute when new columns arrived
   210→   210→    if (cols.length === ds.lastColsLen) return
   211→   211→    ds.lastColsLen = cols.length
   212→   212→
   213→   213→    const inView = cols.filter((c) => c.t >= tMin - 1200)
   214→   214→    // on reset / span change: start from the first in-view column
   215→   215→    if (ds.rawDrawn > inView.length) ds.rawDrawn = 0
   216→   216→
   217→   217→    // consume new raw columns, emitting one visual column per g raw columns
   218→   218→    const newVisual: { rows: [number, { q: number; side: number }][] }[] = []
   219→   219→    while (ds.rawDrawn < inView.length) {
   220→   220→      const col = inView[ds.rawDrawn++]
   221→   221→      for (const side of [0, 1]) {
   222→   222→        const levels = side === 0 ? col.bids : col.asks
   223→   223→        for (const lv of levels) {
   224→   224→          if (lv.p > pTop || lv.p < pBot || lv.q <= 0) continue
   225→   225→          const row = Math.round(lv.p / binSize)
   226→   226→          const not = lv.q * lv.p
   227→   227→          const cur = ds.acc!.rows.get(row)
   228→   228→          if (!cur || not > cur.q * row * binSize) ds.acc!.rows.set(row, { q: lv.q, side })
   229→   229→        }
   230→   230→      }
   231→   231→      ds.acc!.n++
   232→   232→      if (ds.acc!.n >= g) {
   233→   233→        newVisual.push({ rows: [...ds.acc!.rows.entries()] })
   234→   234→        ds.acc!.rows.clear()
   235→   235→        ds.acc!.n = 0
   236→   236→      }
   237→   237→    }
   238→   238→    if (!newVisual.length) return
   239→   239→
   240→   240→    // shift out columns that would scroll off the left edge
   241→   241→    const visibleAfter = ds.vTotal + newVisual.length - ds.shifted
   242→   242→    const needShift = Math.max(0, visibleAfter - G.maxCols)
   243→   243→    if (needShift > 0) {
   244→   244→      ctx.drawImage(hc, -needShift * colW, 0)
   245→   245→      ds.shifted += needShift
   246→   246→    }
   247→   247→
   248→   248→    // intensity reference (median level notional)
   249→   249→    const refQ = aggsRef.current?.refQ ?? 1
   250→   250→
   251→   251→    // draw new visual columns; rightmost sits at the right edge
   252→   252→    const k = newVisual.length
   253→   253→    for (let j = 0; j < k; j++) {
   254→   254→      const x = plotW - (k - j) * colW
   255→   255→      for (const [row, { q, side }] of newVisual[j].rows) {
   256→   256→        const p = row * binSize
   257→   257→        if (p > pTop || p < pBot) continue
   258→   258→        const y = p2yOf(p, pTop, pBot, plotH)
   259→   259→        const h = Math.max(1.5, Math.abs(p2yOf(p - binSize, pTop, pBot, plotH) - y))
   260→   260→        const a = Math.min(0.95, Math.log10(1 + (q * p) / refQ * 3) * 0.5 * contrast)
   261→   261→        if (a <= 0.02) continue
   262→   262→        ctx.fillStyle = side === 0 ? `rgba(${BID_BLUE},${a.toFixed(3)})` : `rgba(${ASK_VIOLET},${a.toFixed(3)})`
   263→   263→        ctx.fillRect(x, Math.floor(y - h / 2), colW + 0.5, Math.ceil(h))
   264→   264→      }
   265→   265→    }
   266→   266→    ds.vTotal += k
   267→   267→  }

   269→   269→  // ------------------------------------------------ main draw loop
   270→   270→  function draw() {
   271→   271→    const el = canvasRef.current
   272→   272→    if (!el || size.w === 0) return
   273→   273→    const ctx = el.getContext('2d')!
   274→   274→    const W = size.w, H = size.h
   275→   275→    const axisW = 62
   276→   276→    const ladderW = showLadder ? 110 : 0
   277→   277→    const padB = 18
   278→   278→    const plotW = W - axisW - ladderW
   279→   279→    const plotH = H - padB
   280→   280→    if (plotW < 60 || plotH < 60) return
   281→   281→
   282→   282→    ctx.fillStyle = BG
   283→   283→    ctx.fillRect(0, 0, W, H)
   284→   284→
   285→   285→    const G = geom(plotW, plotH)
   286→   286→    const tNow = now()
   287→   287→    const tMin = tNow - span
   288→   288→    if (!G || !G.cols.length) {
   289→   289→      ctx.fillStyle = '#8b93a7'
   290→   290→      ctx.font = '12px ui-monospace, monospace'
   291→   291→      ctx.textAlign = 'center'
   292→   292→      ctx.fillText('building local order book…', W / 2, H / 2)
   293→   293→      return
   294→   294→    }
   295→   295→    const { pTop, pBot, binSize, colW, midNow } = G
   296→   296→    const p2y = (p: number) => p2yOf(p, pTop, pBot, plotH)
   297→   297→
   298→   298→    // offscreen depth layer (rebuild when geometry signature changes)
   299→   299→    const sig = `${span}|${vspan}|${binMult}|${contrast}|${plotW}x${plotH}|${G.anchor.toFixed(0)}`
   300→   300→    if (!heatRef.current || sigRef.current !== sig) {
   301→   301→      const dpr = Math.min(2, window.devicePixelRatio || 1)
   302→   302→      const hc = heatRef.current ?? document.createElement('canvas')
   303→   303→      hc.width = Math.max(1, Math.floor(plotW * dpr))
   304→   304→      hc.height = Math.max(1, Math.floor(plotH * dpr))
   305→   305→      const hctx = hc.getContext('2d')!
   306→   306→      hctx.setTransform(dpr, 0, 0, dpr, 0, 0)
   307→   307→      heatRef.current = hc
   308→   308→      sigRef.current = sig
   309→   309→      drawStateRef.current = { vTotal: 0, shifted: 0, rawDrawn: 0, lastColsLen: -1, acc: null }
   310→   310→    }
   311→   311→    const hc = heatRef.current!
   312→   312→    renderDepthInto(hc, G, plotW, plotH)
   313→   313→
   314→   314→    // grid
   315→   315→    ctx.strokeStyle = '#141824'
   316→   316→    ctx.lineWidth = 1
   317→   317→    ctx.font = '10px ui-monospace, monospace'
   318→   318→    const rows = 10
   319→   319→    ctx.textAlign = 'left'
   320→   320→    for (let i = 0; i <= rows; i++) {
   321→   321→      const p = pBot + ((pTop - pBot) * i) / rows
   322→   322→      const y = p2y(p)
   323→   323→      ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(plotW, y); ctx.stroke()
   324→   324→      ctx.fillStyle = '#5b6478'
   325→   325→      ctx.fillText(fmtPrice(p, meta.tick), plotW + 5, y + 3)
   326→   326→    }
   327→   327→
   328→   328→    // 1. depth layer
   329→   329→    ctx.drawImage(hc, 0, 0, plotW, plotH)
   330→   330→
   331→   331→    const aggs = aggsRef.current
   332→   332→    const xOf = (t: number) => ((t - tMin) / span) * plotW
   333→   333→
   334→   334→    // 2. decision lines (nearest big walls)
   335→   335→    if (aggs?.buyAbove) {
   336→   336→      const y = p2y(aggs.buyAbove)
   337→   337→      if (y > 8 && y < plotH - 8) {
   338→   338→        ctx.strokeStyle = 'rgba(240,185,11,0.75)'
   339→   339→        ctx.setLineDash([6, 4]); ctx.lineWidth = 1
   340→   340→        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(plotW, y); ctx.stroke()
   341→   341→        ctx.setLineDash([])
   342→   342→        label(ctx, `SELL BELOW ${fmtPrice(aggs.buyAbove, meta.tick)}`, 6, y - 5, '#f0b90b')
   343→   343→      }
   344→   344→    }
   345→   345→    if (aggs?.sellBelow) {
   346→   346→      const y = p2y(aggs.sellBelow)
   347→   347→      if (y > 8 && y < plotH - 8) {
   348→   348→        ctx.strokeStyle = 'rgba(240,185,11,0.75)'
   349→   349→        ctx.setLineDash([6, 4]); ctx.lineWidth = 1
   350→   350→        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(plotW, y); ctx.stroke()
   351→   351→        ctx.setLineDash([])
   352→   352→        label(ctx, `BUY ABOVE ${fmtPrice(aggs.sellBelow, meta.tick)}`, 6, y + 12, '#f0b90b')
   353→   353→      }
   354→   354→    }
   355→   355→
   356→   356→    // 3. POC
   357→   357→    if (aggs?.poc) {
   358→   358→      const y = p2y(aggs.poc)
   359→   359→      if (y > 4 && y < plotH - 4) {
   360→   360→        ctx.strokeStyle = 'rgba(240,185,11,0.5)'
   361→   361→        ctx.setLineDash([2, 4]); ctx.lineWidth = 1
   362→   362→        ctx.beginPath(); ctx.moveTo(0, y); ctx.lineTo(plotW, y); ctx.stroke()
   363→   363→        ctx.setLineDash([])
   364→   364→        label(ctx, 'POC', plotW - 34, y - 5, '#f0b90b')
   365→   365→      }
   366→   366→    }
   367→   367→
   368→   368→    // 4. trade bubbles
   369→   369→    if (showBubbles && aggs) {
   370→   370→      ctx.lineWidth = 1
   371→   371→      for (const d of aggs.dots) {
   372→   372→        const x = xOf(d.t), y = p2y(d.p)
   373→   373→        if (x < -2 || x > plotW + 2 || y < -2 || y > plotH + 2) continue
   374→   374→        ctx.fillStyle = d.buy ? 'rgba(22,199,132,0.35)' : 'rgba(234,57,67,0.35)'
   375→   375→        ctx.beginPath(); ctx.arc(x, y, 1.2, 0, Math.PI * 2); ctx.fill()
   376→   376→      }
   377→   377→      for (const b of aggs.bubbles) {
   378→   378→        const x = xOf(b.t), y = p2y(b.p)
   379→   379→        if (x < -12 || x > plotW + 12 || y < -12 || y > plotH + 12) continue
   380→   380→        const r = Math.min(15, 1.6 + 1.9 * Math.sqrt(b.n / Math.max(1000, bubbleMin * 4)))
   381→   381→        ctx.beginPath(); ctx.arc(x, y, r, 0, Math.PI * 2)
   382→   382→        ctx.fillStyle = b.buy ? 'rgba(22,199,132,0.45)' : 'rgba(234,57,67,0.45)'
   383→   383→        ctx.fill()
